    \chapter{Nen Tang Toan Hoc va DSP Thuc Te}

    \section{Sensor Fusion (Madgwick IMU)}
    Tren MCU, goc quay duoc cap nhat bang bo loc Madgwick IMU \cite{madgwick2011}. Thuat toan su dung gia toc ke de hieu chinh drift cua gyro va cap nhat quaternion theo buoc thoi gian $\Delta t = 0.005\,\text{s}$ (200 Hz). Quaternion duoc chuan hoa moi chu ky de dam bao tinh truc chuan cua phep quay.

    \subsection{Tien xu ly IMU va quy doi don vi}
    Du lieu thuan tu MPU6050 la gia tri raw (int16) va can duoc chuan hoa ve don vi vat ly.
    Voi thang do gia toc $\pm 4g$ va gyro $\pm 500\,\text{dps}$, ta co:
    \begin{equation}
        a_{x,y,z} = \frac{\text{raw}_{a}}{8192},\quad
        \omega_{x,y,z} = \frac{\text{raw}_{g} - b_g}{65.5} \cdot \frac{\pi}{180}
    \end{equation}
    Trong do $b_g$ la bias gyro duoc uoc luong trong pha calib, va he so quy doi dua ve rad/s phuc vu cap nhat quaternion.

    \subsection{Mo hinh cap nhat Madgwick}
    Madgwick IMU ket hop:
    \begin{itemize}
        \item \textbf{Tich phan gyro} de du doan quaternion.
        \item \textbf{Gradient descent} de dua truot vector gia toc ve thang hang truong trong goc quay.
    \end{itemize}
    Dang tong quat cua dao ham quaternion theo gyro:
    \begin{equation}
        \boldsymbol{\omega}_q = [0,\ \omega_x,\ \omega_y,\ \omega_z]
    \end{equation}
    \begin{equation}
        \dot{\mathbf{q}} = \frac{1}{2}\,\mathbf{q} \otimes \boldsymbol{\omega}_q
    \end{equation}
    Sau do bo sung thanh phan hieu chinh:
    \begin{equation}
        \dot{\mathbf{q}}_{corr} = \dot{\mathbf{q}} - \beta \nabla f(\mathbf{q}, \mathbf{a})
    \end{equation}
    He so $\beta$ dieu chinh muc do tin tuong gia toc ke. Sau moi buoc, quaternion duoc chuan hoa:
    \begin{equation}
        \mathbf{q} \leftarrow \frac{\mathbf{q}}{\lVert \mathbf{q} \rVert}
    \end{equation}

    \subsection{Lam muot quaternion tren host}
    Tren host, quaternion duoc lam muot bang NLERP (normalized linear interpolation) de giam rung goc nho:
    \begin{equation}
        \mathbf{q}_{smooth} = \text{normalize}\big((1-\alpha)\,\mathbf{q}_{prev} + \alpha\,\mathbf{q}_{curr}\big)
    \end{equation}
    NLERP duoc chon thay vi SLERP do chi phi tinh toan thap hon va du muot cho cac thay doi goc nho trong realtime. Gia tri $\alpha$ (mac dinh 0.1) can bang giua do nhay va do on dinh. Viec chuan hoa giup giu quaternion nam tren mat cau don vi, giam sai lech huong nho.

    \subsection{Chuyen quaternion sang yaw/pitch}
    Quaternion giu trang thai quay ben trong nen tranh singularity. Euler yaw/pitch chi duoc su dung lam toa do tra cuu HRIR, khong tham gia cap nhat trang thai, do do rui ro gimbal lock khong tac dong den on dinh noi suy \cite{shoemake1985}.

    \section{Lua chon HRTF va noi suy}
    CSDL HRTF dang CIPIC duoc nap tren host \cite{cipic2001}. He thong chon HRIR theo goc yaw/pitch:
    \begin{itemize}
        \item \textbf{Nearest:} chon tap HRIR gan nhat theo azimuth va elevation.
        \item \textbf{Linear azimuth:} noi suy giua 2 azimuth gan nhat (tuy chon).
    \end{itemize}
    Khac voi phien ban MATLAB trong LaTeX cu, he thong khong su dung gain-weighted panning theo nhieu diem ma chi noi suy 1 chieu theo azimuth de giam chi phi cap nhat.

    \subsection{Mo hinh chon HRIR theo luoi goc}
    Voi mang goc $\{\theta_i\}$ va $\{\phi_j\}$ tu CSDL, ta tim chi so gan nhat:
    \begin{equation}
        i^* = \arg\min_i |\theta_i - \theta|, \quad j^* = \arg\min_j |\phi_j - \phi|
    \end{equation}
    Trong che do noi suy 1 chieu theo azimuth, he thong tim 2 goc ke can $\theta_0, \theta_1$ va he so:
    \begin{equation}
        \alpha = \frac{\theta - \theta_0}{\theta_1 - \theta_0}
    \end{equation}
    Sau do HRIR duoc noi suy tuyen tinh theo tung mau:
    \begin{equation}
        h = (1-\alpha) h_0 + \alpha h_1
    \end{equation}
    Phuong phap nay chua thuc hien can chinh phase hoac do tre (delay alignment), vi vay co the gay spectral coloration nhe trong mot so huong; gioi han nay se duoc ghi nhan o phan Ket luan.


    \section{Block Convolution (Overlap-Add FFT)}
    Xu ly HRTF duoc thuc thi theo block voi overlap-add FFT convolution \cite{oppenheim2010}. Khi HRIR thay doi, FFT cua HRIR duoc tinh truoc va luu lai de giam chi phi trong callback am thanh. Do phuc tap giam tu $\mathcal{O}(NM)$ ve $\mathcal{O}(N\log N)$ cho moi block.

    \subsection{Cau truc overlap-add}
    Voi do dai block $L$ va do dai HRIR $M$, kich thuoc FFT duoc chon la:
    \begin{equation}
        N = 2^{\lceil \log_2(L + M - 1) \rceil}
    \end{equation}
    Moi block $x[n]$ duoc zero-pad len $N$, nhan tren mien tan so voi $H(\omega)$, sau do IFFT de lay $y[n]$. Doan cuoi $N-L$ duoc luu vao buffer overlap va cong vao block tiep theo:
    \begin{equation}
        y_k[n] = \text{IFFT}(\text{FFT}(x_k) \cdot H) + o_{k-1}
    \end{equation}
    Trong do $o_{k-1}$ la overlap tu block truoc. Co che nay giu do tre thap (bang kich thuoc block) va on dinh trong callback realtime.

    \section{Ly thuyet he thong repo dang su dung}
    De tranh nham lan voi ly thuyet cu, phan nay tong hop va phan tich cac khoi thuc te dang duoc su dung trong repo.

    \subsection{Protocol quaternion va dong bo thoi gian}
    Khung du lieu serial co dang:
    \begin{equation}
        q_w, q_x, q_y, q_z, t_{dev}\;\;\text{(ms)}
    \end{equation}
    Host doc tung dong, parse va dua vao hang doi. Khi hang doi day, mau cu bi loai bo de uu tien trang thai moi (last-value wins), giup giam do tre tich luy. Co che nay phu hop voi bai toan head-tracking vi tinh ``cap nhat huong moi nhat'' quan trong hon viec giu day du chuoi mau.

    \subsection{Chuan hoa quaternion va chuyen doi Euler}
    Quaternion duoc chuan hoa de dam bao nam tren mat cau don vi:
    \begin{equation}
        \mathbf{q} \leftarrow \frac{\mathbf{q}}{\lVert \mathbf{q} \rVert}
    \end{equation}
    Yaw/pitch duoc suy ra tu quaternion (ZYX) de tra cuu HRIR:
    \begin{align}
        \text{roll} &= \operatorname{atan2}\big(2(w x + y z),\; 1 - 2(x^2 + y^2)\big) \\
        \text{pitch} &= \arcsin\big(2(w y - z x)\big) \\
        \text{yaw} &= \operatorname{atan2}\big(2(w z + x y),\; 1 - 2(y^2 + z^2)\big)
    \end{align}
    Euler chi dung cho indexing HRTF, con trang thai quay van duoc giu o dang quaternion \cite{shoemake1985}.

    \subsection{NLERP smoothing va chu ky cap nhat}
    De giam rung goc nho, quaternion duoc lam muot bang NLERP:
    \begin{equation}
        \mathbf{q}_{smooth} = \text{normalize}\big((1-\alpha)\mathbf{q}_{prev} + \alpha\mathbf{q}_{curr}\big)
    \end{equation}
    Chu ky cap nhat HRTF $T_{update} = 1/f_{update}$ (mac dinh 50 Hz) giu cho viec doi bo loc theo huong nhin khong lam tang tai callback am thanh.

    \subsection{HRTF selection tu CIPIC}
    HRTF duoc nap tu CIPIC \cite{cipic2001}. Repo chon HRIR theo yaw/pitch bang cach tim goc gan nhat theo elevation va:
    \begin{itemize}
        \item \textbf{Nearest:} chon goc azimuth gan nhat.
        \item \textbf{Linear azimuth:} tim hai goc ke can va noi suy theo he so $\alpha$.
    \end{itemize}
    Trong che do noi suy, danh sach azimuth duoc sap xep tang dan, sau do tim $\theta_0, \theta_1$ va noi suy mau HRIR theo:
    \begin{equation}
        h = (1-\alpha) h_0 + \alpha h_1
    \end{equation}

    \subsection{Overlap-add FFT trong callback}
    Repo dung overlap-add FFT convolution \cite{oppenheim2010}. Kich thuoc FFT:
    \begin{equation}
        N = 2^{\lceil \log_2(L + M - 1) \rceil}
    \end{equation}
    FFT cua HRIR duoc tien tinh, moi block chi can FFT input, nhan phan tu, va IFFT. Phan overlap (do dai $N-L$) duoc luu va cong vao block tiep theo de dam bao tinh lien tuc.

    \subsection{Quan ly buffer am thanh va resampling}
    Audio dau vao duoc doc tu file, gop ve mono (trung binh hai kenh). Neu sample rate khac muc tieu, repo su dung resample polyphase de dua ve $F_s$ mong muon, giup dong bo voi block size va thoi gian callback.

    \subsection{Mo hinh luong va tach thread}
    Serial thread chi lam viec I/O va day queue; orientation worker lam muot quaternion va cap nhat HRIR; audio callback chi thuc thi convolution va output. Cach tach luong nay giu callback ``non-blocking'' va on dinh thoi gian thuc.

    \chapter{Thiet Ke Thoi Gian Thuc}

    \section{Thong so he thong}
    Cac thong so chinh theo repo:
    \begin{itemize}
        \item IMU: 200 Hz
        \item UART: 460800 baud
        \item Audio: 48 kHz, block size 256
        \item HRTF update: 50 Hz
        \item Target perceptual latency: nominally 20--40 ms under stable conditions; co the cao hon tuy OS/driver
    \end{itemize}

    \section{Dong bo timestamp va do tre UART}
    MCU gui quaternion kem timestamp $t_{dev}$ (ms). Khi host nhan du lieu o thoi diem $t_{host}$, do lech dong ho duoc uoc luong boi bo loc EMA:
    \begin{equation}
        \hat{\Delta} \leftarrow (1-\lambda)\,\hat{\Delta} + \lambda\,(t_{host} - t_{dev})
    \end{equation}
    Tu do, do tre UART uoc tinh:
    \begin{equation}
        d_{uart} = t_{host} - (t_{dev} + \hat{\Delta})
    \end{equation}
    Gia tri nay phan anh tre duong truyen va jitter serial. Song song, do tre trong hang doi (queue age) duoc do boi:
    \begin{equation}
        d_{queue} = t_{consume} - t_{host}
    \end{equation}
    Hai dai luong tren duoc thu thap de danh gia su on dinh cua duong truyen IMU -> host.

    \section{Mo hinh do tre end-to-end}
    Tong do tre tu luc IMU cap nhat den khi am thanh ra DAC co the xap xi:
    \begin{equation}
        T_{e2e} = d_{uart} + d_{queue} + T_{update} + T_{proc} + T_{buffer} + T_{dac}
    \end{equation}
    Trong do:
    \begin{itemize}
        \item $T_{update} \approx 1 / f_{update}$ la chu ky cap nhat HRTF (50 Hz -> 20 ms).
        \item $T_{proc}$ la thoi gian convolution trong callback.
        \item $T_{buffer} = L / F_s$ la do tre do block audio.
        \item $T_{dac}$ la do tre phan cung (phu thuoc driver).
    \end{itemize}
    Do tre cam nhan thuc te thuong la gia tri lon nhat theo thoi diem cap nhat (worst-case), do do tan so cap nhat HRTF co vai tro quan trong. Do tre ``staleness'' cua huong nhin co the nam trong:
    \begin{equation}
        T_{stale} \in \big[0,\, 1/f_{update}\big]
    \end{equation}

    \section{Headroom callback va gioi han thoi gian}
    Dieu kien on dinh realtime:
    \begin{equation}
        T_{proc} < T_{buffer}
    \end{equation}
    Dinh nghia headroom:
    \begin{equation}
        H = T_{buffer} - T_{proc}
    \end{equation}
    Neu $H < 0$, he thong se xuat hien underrun. Mot tien ich monitoring thu thap thoi gian callback va so lan overrun de dieu chinh $L$ hoac toi uu FFT.

    \section{On dinh so hoc va bien do}
    FFT convolution su dung float32. De tranh clipping tai output, tin hieu nen duoc dam bao nam trong khoang $[-1, 1]$ bang gain co dinh hoac limiter don gian truoc khi xuat ra driver.

    Voi $F_s = 48000\,\text{Hz}$ va $L = 256$:

    \section{Thuat toan block realtime}
    \begin{algorithm}[H]
    \caption{Realtime Spatial Audio (Repo)}
    \label{alg:repo_pipeline}
    \begin{algorithmic}[1]
    \State Init audio buffer, load CIPIC HRTF
    \State Start Serial thread and Orientation worker
    \While{Audio stream active}
        \State Read mono block $x$ (size $L$)
        \State Fetch current convolver (HRIR FFT precomputed)
        \State $y_L, y_R \gets \text{OverlapAddConvolution}(x, h_{L,R})$
        \State Output $(y_L, y_R)$ via sounddevice
    \EndWhile
    \end{algorithmic}
    \end{algorithm}

    \section{Phan giong nhau}
    \begin{itemize}
        \item Kien truc end-to-end IMU -> UART -> host -> HRTF -> convolution -> output.

    \section{Phan khac biet}
    \begin{itemize}
        \item Fusion chay tren MCU (Madgwick), khong phai MATLAB complementary filter.
        \item Convolution da chuyen sang overlap-add FFT, khong con FIR time-domain.
        \item Sample rate 48 kHz (khong phai 44.1 kHz).
        \item Khong co LR4 crossover va soft clipping trong pipeline hien tai.
        \item Audio driver dung \texttt{sounddevice}, khong su dung ASIO.
    \end{itemize}
